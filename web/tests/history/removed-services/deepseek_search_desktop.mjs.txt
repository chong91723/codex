async function send(text){await page.locator('#prompt').fill(text);await page.locator('#prompt').press('Enter');}
async function finish(id){await page.waitForFunction(id=>state.thread?.id===id&&!state.activeTurns.has(id)&&state.turns.length&&state.turns.at(-1).status!=='inProgress'&&!composerState().pending,id,{timeout:150000});return page.evaluate(()=>state.turns.at(-1));}
async function select(id){const i=await page.evaluate(id=>state.threads.findIndex(t=>t.id===id),id);assert(i>=0);await page.locator('.thread-row[data-thread-id="'+id+'"]').click();await page.waitForFunction(id=>state.thread?.id===id&&!state.threadLoading,id,{timeout:45000});}
async function approval(text,choice){await send(text);const card=page.locator('#requests .request-card').filter({hasText:'Codex 请求你的审批'});await card.first().waitFor({timeout:90000});await card.first().getByRole('button',{name:choice,exact:true}).click();}
